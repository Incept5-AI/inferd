def lru_cache_stats(maxsize):
    def decorator(func):
        cache = OrderedDict()
        stats = {"hits": 0, "misses": 0}

        @wraps(func)
        def wrapper(*args, **kwargs):
            key = (args, tuple(sorted(kwargs.items())))
            if key in cache:
                stats["hits"] += 1
                cache.move_to_end(key)
                return cache[key]
            stats["misses"] += 1
            result = func(*args, **kwargs)
            cache[key] = result
            if len(cache) > maxsize:
                cache.popitem(last=False)
            return result

        wrapper.hits = property(lambda: stats["hits"])
        wrapper.misses = property(lambda: stats["misses"])
        return wrapper

    return decorator
