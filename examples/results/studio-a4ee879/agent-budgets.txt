        cache = OrderedDict()

        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            key = (args, tuple(sorted(kwargs.items())))
            if key in cache:

                cache.move_to_end(key)
                return cache[key]

        wrapper.misses = 0
        return wrapper
