def lru_cache_stats(maxsize):
    def decorator(func):
        cache = OrderedDict()
        hits = 0
        misses = 0

        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            nonlocal hits, misses
            key = (args, tuple(sorted(kwargs.items())))
            if key in cache:
                hits += 1
                cache.move_to_end(key)
                result = cache[key]
            else:
                misses += 1
                result = func(*args, **kwargs)
                cache[key] = result
                if len(cache) > maxsize:
                    cache.popitem(last=False)
            wrapper.hits = hits
            wrapper.misses = misses
            return result

        wrapper.hits = 0
        wrapper.misses = 0
        return wrapper
    return decorator
