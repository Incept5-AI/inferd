import functools
from collections import OrderedDict


def lru_cache_stats(maxsize):
    def decorator(func):
        cache = OrderedDict()

        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            key = (args, tuple(sorted(kwargs.items())))
            if key in cache:
                wrapper.hits += 1
                cache.move_to_end(key)
                return cache[key]
            wrapper.misses += 1
            result = func(*args, **kwargs)
            cache[key] = result
            if len(cache) > maxsize:
                cache.popitem(last=False)
            return result

        wrapper.hits = 0
        wrapper.misses = 0
        return wrapper

    return decorator


# --- usage example ---
@lru_cache_stats(maxsize=2)
def fib(n): return n if n < 2 else fib(n - 1) + fib(n - 2)
print(fib(5), fib(5), f"hits={fib.hits} misses={fib.misses}")
