from functools import wraps
from collections import OrderedDict


def lru_cache_stats(maxsize):
    def decorator(fn):
        cache = OrderedDict()

        @wraps(fn)
        def wrapper(*args, **kwargs):
            key = (args, tuple(sorted(kwargs.items())))
            if key in cache:
                wrapper.hits += 1
                cache.move_to_end(key)
                return cache[key]
            wrapper.misses += 1
            result = fn(*args, **kwargs)
            cache[key] = result
            if len(cache) > maxsize:
                cache.popitem(last=False)
            return result

        wrapper.hits = 0
        wrapper.misses = 0
        return wrapper
    return decorator


# Usage example
@lru_cache_stats(2)
def fib(n): return n if n < 2 else fib(n - 1) + fib(n - 2)
print(fib(5), f"hits={fib.hits} misses={fib.misses}")
