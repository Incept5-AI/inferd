        misses = 0

        @functools.wraps(func)

            nonlocal hits, misses
            key = (args, tuple(sorted(kwargs.items())))
            if key in cache:
                hits += 1
                cache.move_to_end(key)

            return result

        wrapper.misses = property(lambda: misses)
        return wrapper
